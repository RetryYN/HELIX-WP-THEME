 * @package THEMEA:R

 * カスタマイザーのパネルとセクションを呼び出す
 */
$felt_panels = array(
	'site-setting',
	'design-preset-setting',
	'site-design-setting',
	'color-setting',
	'main-visual-setting',
	'headline-design-setting',
	'box-design-setting',
	'button-design-setting',
	'fukidashi-setting',
	'profile-setting',
	'information-setting',
	'thumbnails-setting',
	'sns-setting',
	'spmenu-setting',
	'animation-setting',
	'others-setting',
	'representation-act-setting',
);

foreach ($felt_panels as $p) {
	require THEMEA_PHP_INCLUDE . 'customizer/ui/' . $p . '.php';
}


/**
 * カスタマイザーに任意のCSSを読み込ませる
 */

function themeA_customizer_css()
{
	wp_enqueue_style('themeA-block', get_template_directory_uri() . '/block.css');
	wp_enqueue_style('themeA-customizer-css', get_template_directory_uri() . '/themeA-customizer.css');
	if (has_filter('use_widgets_block_editor')) {
		wp_enqueue_style('themeA-block', get_template_directory_uri() . '/block.css');
	}
}
add_action('customize_controls_enqueue_scripts', 'themeA_customizer_css');

/**
 * カスタマイザーに任意のCSSをインラインで読み込ませる
 */
function themeA_customizer_inline_css()
{
	$theme_color              = themeA__theme_color();
	$text_color               = themeA__text_color();
	$theme_color_rgba 		  = themeA_hex_to_rgb($theme_color);
	$text_color_rgba 		  = themeA_hex_to_rgb($text_color);
?>
