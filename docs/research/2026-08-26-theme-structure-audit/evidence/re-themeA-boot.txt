/**
 * jQueryをフッターで読み込み
 */
function themeA_jquery_enqueue()
{
	if (!is_admin()) {
		global $wp_scripts;
		$jquery     = $wp_scripts->registered['jquery-core'];
		$jquery_ver = $jquery->ver;
		$jquery_src = $jquery->src;

		wp_deregister_script('jquery');
		wp_deregister_script('jquery-core');

		wp_register_script('jquery', false, array('jquery-core'), $jquery_ver, true);
		wp_register_script('jquery-core', $jquery_src, array(), $jquery_ver, true);
		wp_enqueue_script('jquery');
		wp_enqueue_script('jquery-core');
	}
}
add_action('wp_enqueue_scripts', 'themeA_jquery_enqueue');


/**
 * jsの読み込み(js output)
 */
function themeA_enqueue_script()
{
	global $post;
	wp_enqueue_script('themeA-youtube-script', get_template_directory_uri() . '/lib/js/youtube.js', array('jquery'), false, true);
	if (empty($post)) {
		wp_register_script('themeA-common-script', get_template_directory_uri() . '/lib/js/common.js', array('jquery'), false, true);
		$common_data = array(
			'ajax_url' => admin_url('admin-ajax.php'),
			'action'   => 'themeA_increment_views'
		);
		wp_localize_script('themeA-common-script', 'themeA_ajax_common', $common_data);
		wp_enqueue_script('themeA-common-script');
		wp_localize_script('themeA-common-script', 'headerTracking', array(
			'tracking_info' => themeA__header_tracking(),
		));
	} else {
		wp_register_script('themeA-common-script', get_template_directory_uri() . '/lib/js/common.js', array('jquery'), false, true);
		$common_data = array(
			'ajax_url' => admin_url('admin-ajax.php'),
			'post_id'  => $post->ID,
			'action'   => 'themeA_increment_views'
		);
		wp_localize_script('themeA-common-script', 'themeA_ajax_common', $common_data);
		wp_enqueue_script('themeA-common-script');
		wp_localize_script('themeA-common-script', 'headerTracking', array(
			'tracking_info' => themeA__header_tracking(),
		));
	}

	// スライダーJSは記事スライドもしくは画像スライドのスライドインの時のみ読み込み
	if (themeA__main_visual_type() == 'type01-post-slider') {
		if (themeA__slider_design_select() == 'd--slider-design1') {
			wp_enqueue_script('themeA-slider-script', get_template_directory_uri() . '/lib/js/slider.js', array('jquery'), false, true);
		} else {
			wp_enqueue_style('themeA-swiper-style', 'https://cdn.jsdelivr.net/npm/swiper@8/swiper-bundle.min.css');
			wp_enqueue_script('themeA-swiper-script', 'https://cdn.jsdelivr.net/npm/swiper@8/swiper-bundle.min.js', array(), false, false);
			wp_enqueue_script('themeA-swiper-setting-script', get_template_directory_uri() . '/lib/js/slider-jin.js', array(), false, true);
		}
	} elseif (themeA__main_visual_type() == 'type02-image-slider'  && themeA__imageslider_animation_select() == 'd--imageslider-animation-slidein') {
		wp_enqueue_script('themeA-slider-script', get_template_directory_uri() . '/lib/js/slider.js', array('jquery'), false, true);
	}
	if (themeA__add_class() !== false) {
		wp_enqueue_script('add-themeA-class', get_template_directory_uri() . '/lib/js/add-themeA-class.js', array('jquery'), false, true);
	}
	wp_enqueue_script('themeA-tab-switch-script', get_template_directory_uri() . '/lib/js/tab_switch.js', array('jquery'), false, true);
}
add_action('wp_enqueue_scripts', 'themeA_enqueue_script');
